# set conda environment
## unload python as Conda uses Python but prefers its own installation
module unload python
## create rna-seq environment 
conda create -p /N/slate/yl278/rna-seq \
  -c conda-forge -c bioconda \
  fastqc cutadapt star samtools subread
## To activate this environment, use                                            
conda activate /N/slate/yl278/rna-seq                                                 
## To deactivate an active environment, use                                     
conda deactivate  

# get RNA-seq data (terminal):
cd /N/project/HundleyLab  \
ls \
cd GSFxxxx — Replace `xxxx` with your RNA-seq dataset number (for example, `4595`) \
ls - output is like 'archive-GSF4595.run1.tar'  \
mkdir -p /N/slate/yl278/GSFxxxx
cp archive-GSF4595.run1.tar /N/slate/yl278/GSFxxxx \
## List the files in the archive (head 40) to check whether these files are what we want
tar -tf /N/slate/yl278/GSF4595/archive-GSF4595.run1.tar | head -40
## Extract to a separate original data directory
mkdir -p /N/slate/yl278/GSF4595/raw_data
tar -xvf /N/slate/yl278/GSF4595/archive-GSF4595.run1.tar \
  -C /N/slate/yl278/GSF4595/raw_data
